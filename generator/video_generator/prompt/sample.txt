            UP * 0.5,  # root

            RIGHT * 1 + DOWN * 3,  # level 2 right-middle

            circles.add(circle)
            labels.add(label)

        # Insert words one by one
        current_text = Text("Inserting words:", font_size=20, color=GREEN)
        current_text.to_edge(DOWN)
        self.play(Write(current_text))
        
        # Node storage for dynamic creation
        nodes = {"": root_circle}  # prefix -> node
        node_labels = {"": root_label}
        edges = []
        
        for word_idx, word in enumerate(words):
            word_text = Text(f"Insert: {word}", font_size=18, color=ORANGE)
            word_text.to_edge(DOWN).shift(UP * 0.5)
            self.play(Write(word_text))
            
            current_prefix = ""
            current_node = root_circle
            
            for char_idx, char in enumerate(word):
                new_prefix = current_prefix + char
                
                if new_prefix not in nodes:
                    # Create new node
                    new_circle = Circle(radius=0.2, color=WHITE, fill_opacity=0.3)
                    
                    # Position calculation
                    level = char_idx + 1
                    branch_offset = (ord(char) - ord('A')) * 0.8 - 2
                    
                    x_pos = branch_offset * (0.8 ** level)
                    y_pos = 1.5 - level * 0.8
                    
                    new_circle.move_to(RIGHT * x_pos + UP * y_pos)
                    
                    # Create label
                    char_label = Text(char, font_size=12, color=BLACK)
                    char_label.move_to(new_circle.get_center())
                    
                    # Create edge
                    edge = Line(current_node.get_center(), new_circle.get_center(), color=YELLOW)
                    
                    # Animate creation
                    self.play(Create(edge), Create(new_circle), Write(char_label))
                    
                    nodes[new_prefix] = new_circle
                    node_labels[new_prefix] = char_label
                    edges.append(edge)
                    
                    self.wait(0.3)
                
                current_node = nodes[new_prefix]
                current_prefix = new_prefix
            
            # Mark end of word
            end_marker = Text("*", font_size=16, color=RED)
            end_marker.move_to(current_node.get_center() + DOWN * 0.3)
            self.play(Write(end_marker))
            
            self.play(FadeOut(word_text))
            self.wait(0.5)
        
        self.wait(2)
        
        # Search operation
        self.play(FadeOut(current_text))
        search_text = Text("Search: CAR", font_size=24, color=GREEN)
        search_text.to_edge(DOWN)
        self.play(Write(search_text))

        search_word = "CAR"
        search_path = [""]
        
        for char in search_word:

from manim import *

class SegmentTreeVisualization(Scene):
    def construct(self):
        # Title
        title = Text("Segment Tree", font_size=48, color=BLUE)
        title.to_edge(UP)
        self.play(Write(title))
        self.wait(1)
        
        # Purpose
        purpose_text = Text("Efficient Range Queries and Updates", font_size=20, color=YELLOW)
        purpose_text.shift(UP * 2.5)
        self.play(Write(purpose_text))
        self.wait(1)
        
        # Original array
        array_values = [1, 3, 5, 7, 9, 11]
        array_text = Text("Array: [1, 3, 5, 7, 9, 11]", font_size=18, color=GREEN)
        array_text.shift(UP * 2)
        self.play(Write(array_text))
        self.wait(1)
        
        # Build segment tree (sum tree)
        build_text = Text("Building Segment Tree (Sum)", font_size=20, color=ORANGE)
        build_text.to_edge(DOWN)
        self.play(Write(build_text))
        
        # Tree structure for sum segment tree
        # Tree values: [36, 9, 27, 4, 5, 16, 11, 1, 3, 0, 0, 7, 9, 0, 0]
        tree_values = [36, 9, 27, 4, 5, 16, 11, 1, 3, 0, 0, 7, 9, 0, 0]
        
        # Create nodes (only non-zero values)
        nodes = VGroup()
        node_labels = VGroup()
        lines = VGroup()
        
        # Node positions
        positions = [
            UP * 0.5,                    # root (36)
            LEFT * 2 + DOWN * 1,         # left child (9)
            RIGHT * 2 + DOWN * 1,        # right child (27)
            LEFT * 3 + DOWN * 2,         # (4)
            LEFT * 1 + DOWN * 2,         # (5)
            RIGHT * 1 + DOWN * 2,        # (16)
            RIGHT * 3 + DOWN * 2,        # (11)
            LEFT * 3.5 + DOWN * 3,       # (1)
            LEFT * 2.5 + DOWN * 3,       # (3)
            RIGHT * 0.5 + DOWN * 3,      # (7)
            RIGHT * 1.5 + DOWN * 3,      # (9)
        ]
        
        displayed_values = [36, 9, 27, 4, 5, 16, 11, 1, 3, 7, 9]
        
        # Create nodes
        for i, (value, pos) in enumerate(zip(displayed_values, positions)):
            circle = Circle(radius=0.25, color=WHITE, fill_opacity=0.3)
            circle.move_to(pos)
            label = Text(str(value), font_size=12)
            label.move_to(circle.get_center())
            
            nodes.add(circle)
            node_labels.add(label)
        
        # Create connections
        connections = [
            (0, 1), (0, 2),  # root to children
            (1, 3), (1, 4),  # left subtree
            (2, 5), (2, 6),  # right subtree
            (3, 7), (3, 8),  # leaf level
            (5, 9), (5, 10)
        ]
        
        for parent, child in connections:
            line = Line(nodes[parent].get_center(), nodes[child].get_center(), color=YELLOW)
            lines.add(line)
        
        # Animate tree construction
        self.play(Create(nodes), Write(node_labels))
        self.play(Create(lines))
        self.wait(2)
        
        # Range query
        self.play(FadeOut(build_text))
        query_text = Text("Range Query: Sum(1, 3) = ?", font_size=24, color=GREEN)
        query_text.to_edge(DOWN)
        self.play(Write(query_text))
        
        # Highlight nodes involved in query
        query_nodes = [0, 1, 2, 4, 5, 9]  # nodes that contribute to sum(1,3)
        highlights = VGroup()
        
        for node_idx in [1, 4, 9]:  # key nodes for range [1,3]
            highlight = Circle(radius=0.35, color=GREEN, stroke_width=4)
            highlight.move_to(nodes[node_idx].get_center())
            highlights.add(highlight)
        
        self.play(Create(highlights))
        
        # Show calculation
        calc_text = Text("5 + 7 + 9 = 21", font_size=18, color=GREEN)
        calc_text.to_edge(DOWN).shift(UP * 0.5)
        self.play(Write(calc_text))
        self.wait(2)
        
        # Update operation
        self.play(FadeOut(query_text), FadeOut(calc_text), FadeOut(highlights))
        update_text = Text("Update: Change index 2 from 5 to 6", font_size=20, color=ORANGE)
        update_text.to_edge(DOWN)
        self.play(Write(update_text))
        
        # Show update propagation
        update_path = [4, 1, 0]  # path from leaf to root
        update_highlights = VGroup()
        
        for node_idx in update_path:
            highlight = Circle(radius=0.35, color=ORANGE, stroke_width=4)
            highlight.move_to(nodes[node_idx].get_center())
            update_highlights.add(highlight)
        
        self.play(Create(update_highlights))
        
        # Update values
        new_values = [6, 10, 37]  # new values after update
        for i, (node_idx, new_val) in enumerate(zip(update_path, new_values)):
            new_label = Text(str(new_val), font_size=12)
            new_label.move_to(nodes[node_idx].get_center())
            self.play(ReplacementTransform(node_labels[node_idx], new_label))
            node_labels[node_idx] = new_label
        
        self.wait(2)
        
        # Time complexity
        self.play(FadeOut(update_text), FadeOut(update_highlights))
        complexity_title = Text("Time Complexity:", font_size=20, color=PURPLE)
        complexity_title.to_edge(DOWN).shift(UP * 1)
        
        complexity_list = VGroup(
            Text("• Build: O(n)", font_size=16),
            Text("• Query: O(log n)", font_size=16),
            Text("• Update: O(log n)", font_size=16)
        )
        
        complexity_list.arrange(DOWN, aligned_edge=LEFT)
        complexity_list.to_edge(DOWN)
        
        self.play(Write(complexity_title))
        self.play(Write(complexity_list))
        self.wait(3)
        
        # Applications
        apps_text = Text("Applications: Range sum, Range minimum/maximum, Range updates", 
                        font_size=16, color=YELLOW)
        apps_text.to_edge(DOWN).shift(DOWN * 0.5)
        self.play(Write(apps_text))
        self.wait(3)
