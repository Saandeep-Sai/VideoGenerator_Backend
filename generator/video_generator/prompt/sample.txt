# PROFESSIONAL MANIM SAMPLES - Safe Positioning + Engaging Animations

from manim import *

config.frame_width = 16
config.frame_height = 9
config.pixel_width = 1920
config.pixel_height = 1080

# 
# EXAMPLE 1: Dynamic Title with Perfect Spacing
# 
class DynamicIntro(Scene):
    def construct(self):
        # Title in TOP zone
        title = Text("The Future of AI", font_size=56, weight=BOLD, color=BLUE)
        title.scale_to_fit_width(config.frame_width * 0.85)  #  MANDATORY
        title.move_to(UP * 2.5)  #  Safe TOP position
        
        self.play(Write(title, run_time=1.5))
        self.play(title.animate.set_color(GOLD), Circumscribe(title, color=YELLOW))
        
        # Subtitle in MIDDLE zone (2.5 units from title)
        subtitle = Text("Transforming Tomorrow", font_size=38, color=WHITE)
        subtitle.scale_to_fit_width(config.frame_width * 0.8)
        subtitle.move_to(ORIGIN)  #  Safe MIDDLE
        
        self.play(GrowFromCenter(subtitle))
        self.play(Flash(subtitle, color=BLUE))
        
        # Call to action in BOTTOM zone
        cta = Text("Let's Explore", font_size=32, color=GREEN)
        cta.scale_to_fit_width(config.frame_width * 0.7)
        cta.move_to(DOWN * 2.5)  #  Safe BOTTOM
        
        self.play(FadeIn(cta, shift=UP*0.3))
        self.play(Indicate(cta, scale_factor=1.2))
        
        self.play(Uncreate(title), FadeOut(subtitle, scale=0.5), FadeOut(cta))


# 
# EXAMPLE 2: Bullet List with Cascade Effect
# 
class EngagingList(Scene):
    def construct(self):
        title = Text("Key Benefits", font_size=48, weight=BOLD, color=BLUE)
        title.scale_to_fit_width(config.frame_width * 0.85)
        title.move_to(UP * 2.8)
        self.play(DrawBorderThenFill(title))
        
        point1 = Text(" Speed: 10x faster", font_size=34, color=WHITE)
        point1.scale_to_fit_width(config.frame_width * 0.75)
        point1.move_to(UP * 1.2)
        
        point2 = Text(" Accuracy: 99.9%", font_size=34, color=WHITE)
        point2.scale_to_fit_width(config.frame_width * 0.75)
        point2.move_to(ORIGIN)
        
        point3 = Text(" Efficiency: 50% savings", font_size=34, color=WHITE)
        point3.scale_to_fit_width(config.frame_width * 0.75)
        point3.move_to(DOWN * 1.2)
        
        points = VGroup(point1, point2, point3)
        self.play(AnimationGroup(*[Write(p) for p in points], lag_ratio=0.3))
        
        for point in points:
            self.play(Flash(point, color=YELLOW), point.animate.set_color(GOLD))
        
        self.play(FadeOut(title, shift=UP), *[FadeOut(p, shift=LEFT*0.5) for p in points])


# 
# KEY RULES:
# 
#  ALWAYS: .scale_to_fit_width(config.frame_width * 0.85) on ALL text
#  Safe zones: TOP (UP*2.5-3.0), MIDDLE (0.5), BOTTOM (DOWN*2.5)
#  Spacing: Minimum 1.5 units between objects
#  Bounds: -6.5 to +6.5 horizontal, -3.5 to +3.5 vertical
#  Diversity: Use 8+ different animations (Write, GrowFromCenter, Flash, etc.)
