# Script 3: Stack Data Structure
from manim import *

class StackVisualization(Scene):
    def construct(self):
        # Title
        title = Text("Stack Data Structure (LIFO)", font_size=48, color=BLUE)
        title.to_edge(UP)
        self.play(Write(title))
        self.wait(1)
        
        # Create stack base
        stack_base = Rectangle(width=2, height=0.3, color=BROWN, fill_opacity=0.8)
        stack_base.shift(DOWN * 2.5)
        self.play(Create(stack_base))
        
        # Stack operations
        stack_elements = []
        values = [10, 20, 30, 40, 50]
        
        # Push operations
        push_text = Text("PUSH Operations", font_size=24, color=GREEN)
        push_text.to_edge(DOWN)
        self.play(Write(push_text))
        
        for i, value in enumerate(values):
            # Create stack element
            element = Rectangle(width=1.8, height=0.5, color=WHITE, fill_opacity=0.3)
            element.shift(DOWN * (2 - i * 0.6))
            
            # Add value label
            label = Text(str(value), font_size=20)
            label.move_to(element.get_center())
            
            # Push animation
            element.shift(UP * 3)
            label.shift(UP * 3)
            
            self.play(Create(element), Write(label))
            self.play(element.animate.shift(DOWN * 3), label.animate.shift(DOWN * 3))
            
            stack_elements.append((element, label))
            
            # Show top pointer
            if i == 0:
                top_arrow = Arrow(
                    start=element.get_center() + LEFT * 1.5,
                    end=element.get_center() + LEFT * 1,
                    buff=0,
                    color=RED,
                    stroke_width=4
                )
                top_text = Text("TOP", font_size=16, color=RED)
                top_text.move_to(element.get_center() + LEFT * 2)
                self.play(Create(top_arrow), Write(top_text))
            else:
                # Move top pointer
                self.play(
                    top_arrow.animate.shift(UP * 0.6),
                    top_text.animate.shift(UP * 0.6)
                )
            
            self.wait(0.5)
        
        self.wait(2)
        
        # Pop operations
        self.play(FadeOut(push_text))
        pop_text = Text("POP Operations", font_size=24, color=ORANGE)
        pop_text.to_edge(DOWN)
        self.play(Write(pop_text))
        
        for i in range(2):  # Pop 2 elements
            if stack_elements:
                element, label = stack_elements.pop()
                
                # Pop animation
                self.play(
                    element.animate.shift(UP * 3),
                    label.animate.shift(UP * 3)
                )
                self.play(FadeOut(element), FadeOut(label))
                
                # Move top pointer down
                if stack_elements:
                    self.play(
                        top_arrow.animate.shift(DOWN * 0.6),
                        top_text.animate.shift(DOWN * 0.6)
                    )
                
                self.wait(0.5)
        
        self.wait(2)
        
        # Stack applications
        self.play(FadeOut(pop_text))
        apps_title = Text("Stack Applications:", font_size=20, color=YELLOW)
        apps_title.to_edge(DOWN).shift(UP * 1.5)
        
        apps_list = VGroup(
            Text("• Function call management", font_size=16),
            Text("• Expression evaluation", font_size=16),
            Text("• Undo operations", font_size=16),
            Text("• Browser history", font_size=16)
        )
        
        apps_list.arrange(DOWN, aligned_edge=LEFT)
        apps_list.to_edge(DOWN)
        
        self.play(Write(apps_title))
        self.play(Write(apps_list))
        self.wait(3)
        
        # Time complexity
        complexity_text = Text("Time Complexity: Push O(1), Pop O(1), Peek O(1)", 
                             font_size=18, color=PURPLE)
        complexity_text.to_edge(DOWN).shift(UP * 0.3)
        self.play(Write(complexity_text))
        self.wait(3)

class QueueVisualization(Scene):
    def construct(self):
        # Title
        title = Text("Queue Data Structure (FIFO)", font_size=48, color=BLUE)
        title.to_edge(UP)
        self.play(Write(title))
        self.wait(1)
        
        # Create queue container

        queue_container.shift(DOWN * 0.5)
        self.play(Create(queue_container))
        
        # Front and rear pointers
        front_text = Text("FRONT", font_size=16, color=GREEN)
        rear_text = Text("REAR", font_size=16, color=RED)
        
        front_text.move_to(queue_container.get_left() + RIGHT * 0.5 + UP * 1)
        rear_text.move_to(queue_container.get_left() + RIGHT * 0.5 + DOWN * 1)
        
        front_arrow = Arrow(
            start=front_text.get_center() + DOWN * 0.3,
            end=queue_container.get_left() + RIGHT * 0.5,
            buff=0,
            color=GREEN,
            stroke_width=3
        )
        
        rear_arrow = Arrow(
            start=rear_text.get_center() + UP * 0.3,
            end=queue_container.get_left() + RIGHT * 0.5,
            buff=0,
            color=RED,
            stroke_width=3
        )
        
        self.play(Write(front_text), Write(rear_text))
        self.play(Create(front_arrow), Create(rear_arrow))
        self.wait(1)
        
        # Enqueue operations
        enqueue_text = Text("ENQUEUE Operations", font_size=24, color=GREEN)
        enqueue_text.to_edge(DOWN)
        self.play(Write(enqueue_text))
        
        queue_elements = []
        values = [10, 20, 30, 40, 50]
        
        for i, value in enumerate(values):
            # Create queue element
            element = Rectangle(width=1.4, height=1.2, color=WHITE, fill_opacity=0.3)
            element.move_to(queue_container.get_left() + RIGHT * (0.8 + i * 1.5))
            
            # Add value label
            label = Text(str(value), font_size=18)
            label.move_to(element.get_center())
            
            # Enqueue animation from bottom
            element.shift(DOWN * 3)
            label.shift(DOWN * 3)
            
            self.play(Create(element), Write(label))
            self.play(element.animate.shift(UP * 3), label.animate.shift(UP * 3))
            
            queue_elements.append((element, label))
            
            # Move rear pointer
            self.play(
                rear_arrow.animate.shift(RIGHT * 1.5),
                rear_text.animate.shift(RIGHT * 1.5)
            )
            
            self.wait(0.5)
        
        self.wait(2)
        
        # Dequeue operations
        self.play(FadeOut(enqueue_text))
        dequeue_text = Text("DEQUEUE Operations", font_size=24, color=ORANGE)
        dequeue_text.to_edge(DOWN)
        self.play(Write(dequeue_text))
        
        for i in range(2):  # Dequeue 2 elements
            if queue_elements:
                element, label = queue_elements.pop(0)
                
                # Dequeue animation
                self.play(
                    element.animate.shift(UP * 3),
                    label.animate.shift(UP * 3)
                )
                self.play(FadeOut(element), FadeOut(label))
                
                # Move front pointer
                self.play(
                    front_arrow.animate.shift(RIGHT * 1.5),
                    front_text.animate.shift(RIGHT * 1.5)
                )
                
                self.wait(0.5)
        
        self.wait(2)
        
        # Queue applications
        self.play(FadeOut(dequeue_text))
        apps_title = Text("Queue Applications:", font_size=20, color=YELLOW)
        apps_title.to_edge(DOWN).shift(UP * 1.5)
        
        apps_list = VGroup(
            Text("• Process scheduling", font_size=16),
            Text("• BFS algorithm", font_size=16),
            Text("• Print job management", font_size=16),
            Text("• Buffer for data streams", font_size=16)
        )
        
        apps_list.arrange(DOWN, aligned_edge=LEFT)
        apps_list.to_edge(DOWN)
        
        self.play(Write(apps_title))
        self.play(Write(apps_list))
        self.wait(3)

        self.play(Write(complexity_text))

from manim import *

class BinarySearchTree(Scene):

        self.play(Write(property_text))
        self.wait(1)
        
        # Create BST nodes
        root_circle = Circle(radius=0.3, color=WHITE, fill_opacity=0.3)
        root_label = Text("50", font_size=16)
        root_label.move_to(root_circle.get_center())
        root_node = VGroup(root_circle, root_label)
        
        # Left subtree
        left_circle = Circle(radius=0.3, color=WHITE, fill_opacity=0.3)
        left_circle.shift(LEFT * 2 + DOWN * 1.5)
        left_label = Text("30", font_size=16)
        left_label.move_to(left_circle.get_center())
        left_node = VGroup(left_circle, left_label)
        
        # Right subtree
        right_circle = Circle(radius=0.3, color=WHITE, fill_opacity=0.3)
        right_circle.shift(RIGHT * 2 + DOWN * 1.5)
        right_label = Text("70", font_size=16)
        right_label.move_to(right_circle.get_center())
        right_node = VGroup(right_circle, right_label)
        
        # Create connections
        left_line = Line(root_circle.get_center(), left_circle.get_center(), color=YELLOW)
        right_line = Line(root_circle.get_center(), right_circle.get_center(), color=YELLOW)
        
        # Animate tree creation
        self.play(Create(root_node))
        self.wait(0.5)
        
        self.play(Create(left_line), Create(left_node))
        self.wait(0.5)
        
        self.play(Create(right_line), Create(right_node))
        self.wait(1)
        
        # Add more nodes
        nodes_data = [
            (20, LEFT * 3 + DOWN * 3, left_circle),
            (40, LEFT * 1 + DOWN * 3, left_circle),
            (60, RIGHT * 1 + DOWN * 3, right_circle),
            (80, RIGHT * 3 + DOWN * 3, right_circle)
        ]
        
        nodes = [root_node, left_node, right_node]
        lines = [left_line, right_line]
        
        for value, position, parent in nodes_data:
            circle = Circle(radius=0.3, color=WHITE, fill_opacity=0.3)
            circle.move_to(position)
            label = Text(str(value), font_size=16)
            label.move_to(circle.get_center())
            node = VGroup(circle, label)
            
            line = Line(parent.get_center(), circle.get_center(), color=YELLOW)
            
            self.play(Create(line), Create(node))
            nodes.append(node)
            lines.append(line)
            self.wait(0.3)
        
        self.wait(2)
        
        # Search operation
        search_text = Text("Search for 40", font_size=24, color=GREEN)
        search_text.to_edge(DOWN)
        self.play(Write(search_text))
        
        # Highlight search path
        search_path = [0, 1, 5]  # indices of nodes in search path

            highlight.move_to(nodes[i].get_center())
            self.play(Create(highlight))
            self.wait(0.8)
            if i != search_path[-1]:
                self.play(FadeOut(highlight))
        
        self.wait(2)
        
        # Insert operation
        self.play(FadeOut(search_text))
        insert_text = Text("Insert 35", font_size=24, color=GREEN)
        insert_text.to_edge(DOWN)
        self.play(Write(insert_text))

            highlight = Circle(radius=0.4, color=BLUE, stroke_width=4)
            highlight.move_to(nodes[i].get_center())
            self.play(Create(highlight))
            self.wait(0.5)
            self.play(FadeOut(highlight))

        new_circle = Circle(radius=0.3, color=WHITE, fill_opacity=0.3)
        new_circle.shift(LEFT * 0.5 + DOWN * 4.5)
        new_label = Text("35", font_size=16)

# Script 6: Hash Table
from manim import *

class HashTableVisualization(Scene):
    def construct(self):
        # Title
        title = Text("Hash Table Data Structure", font_size=48, color=BLUE)
        title.to_edge(UP)
        self.play(Write(title))
        self.wait(1)
        
        # Hash function explanation
        hash_text = Text("Hash Function: key % table_size", font_size=20, color=YELLOW)
        hash_text.shift(UP * 2.5)
        self.play(Write(hash_text))
        self.wait(1)
        
        # Create hash table
        table_size = 7
        table_boxes = VGroup()
        index_labels = VGroup()
        
        for i in range(table_size):
            box = Rectangle(width=1.5, height=0.8, color=WHITE, fill_opacity=0.2)
            box.shift(DOWN * (i - 3) * 1.1)
            
            index_label = Text(str(i), font_size=18, color=GRAY)
            index_label.move_to(box.get_left() + LEFT * 0.5)
            
            table_boxes.add(box)
            index_labels.add(index_label)
        
        self.play(Create(table_boxes))
        self.play(Write(index_labels))
        self.wait(1)
        
        # Hash table operations
        keys_values = [(22, "Apple"), (15, "Banana"), (29, "Cherry"), (8, "Date")]
        
        for key, value in keys_values:
            # Show hash calculation
            calc_text = Text(f"Hash({key}) = {key} % {table_size} = {key % table_size}", 
                           font_size=18, color=GREEN)
            calc_text.to_edge(DOWN)
            self.play(Write(calc_text))
            
            # Calculate hash index
            hash_index = key % table_size
            
            # Highlight target box
            highlight = Rectangle(width=1.5, height=0.8, color=RED, stroke_width=4)
            highlight.move_to(table_boxes[hash_index].get_center())
            self.play(Create(highlight))
            
            # Insert key-value pair
            kv_text = Text(f"{key}:{value}", font_size=12)
            kv_text.move_to(table_boxes[hash_index].get_center())
            self.play(Write(kv_text))
            
            self.play(FadeOut(highlight))

class TrieVisualization(Scene):
    def construct(self):

        self.play(Write(definition))
        self.wait(1)
        
        # Words to insert
        words = ["CAT", "CAR", "CARD", "CARE", "CAREFUL"]
        
        # Create root node
        root_circle = Circle(radius=0.2, color=WHITE, fill_opacity=0.3)
        root_circle.shift(UP * 1.5)
        root_label = Text("ROOT", font_size=10)
        root_label.move_to(root_circle.get_center() + DOWN * 0.5)
        
        self.play(Create(root_circle), Write(root_label))
        self.wait(1)
        
        # Insert words one by one
        current_text = Text("Inserting words:", font_size=20, color=GREEN)
        current_text.to_edge(DOWN)
        self.play(Write(current_text))
        
        # Node storage for dynamic creation
        nodes = {"": root_circle}  # prefix -> node
        node_labels = {"": root_label}
        edges = []
        
        for word_idx, word in enumerate(words):
            word_text = Text(f"Insert: {word}", font_size=18, color=ORANGE)
            word_text.to_edge(DOWN).shift(UP * 0.5)
            self.play(Write(word_text))
            
            current_prefix = ""
            current_node = root_circle
            
            for char_idx, char in enumerate(word):
                new_prefix = current_prefix + char
                
                if new_prefix not in nodes:
                    # Create new node
                    new_circle = Circle(radius=0.2, color=WHITE, fill_opacity=0.3)
                    
                    # Position calculation
                    level = char_idx + 1
                    branch_offset = (ord(char) - ord('A')) * 0.8 - 2
                    
                    x_pos = branch_offset * (0.8 ** level)
                    y_pos = 1.5 - level * 0.8
                    
                    new_circle.move_to(RIGHT * x_pos + UP * y_pos)
                    
                    # Create label
                    char_label = Text(char, font_size=12, color=BLACK)
                    char_label.move_to(new_circle.get_center())
                    
                    # Create edge
                    edge = Line(current_node.get_center(), new_circle.get_center(), color=YELLOW)
                    
                    # Animate creation
                    self.play(Create(edge), Create(new_circle), Write(char_label))
                    
                    nodes[new_prefix] = new_circle
                    node_labels[new_prefix] = char_label
                    edges.append(edge)
                    
                    self.wait(0.3)
                
                current_node = nodes[new_prefix]
                current_prefix = new_prefix
            
            # Mark end of word
            end_marker = Text("*", font_size=16, color=RED)
            end_marker.move_to(current_node.get_center() + DOWN * 0.3)
            self.play(Write(end_marker))
            
            self.play(FadeOut(word_text))
            self.wait(0.5)
        
        self.wait(2)
        
        # Search operation
        self.play(FadeOut(current_text))
        search_text = Text("Search: CAR", font_size=24, color=GREEN)
        search_text.to_edge(DOWN)
        self.play(Write(search_text))
        
        # Highlight search path
        search_word = "CAR"
        search_path = [""]
        
        for char in search_word:
            search_path.append(search_path[-1] + char)
        
        # Animate search
        for i in range(1, len(search_path)):
            prefix = search_path[i]
            if prefix in nodes:
                highlight = Circle(radius=0.3, color=GREEN, stroke_width=4)
                highlight.move_to(nodes[prefix].get_center())

                self.wait(0.5)
