class Segment000(Scene):
    def construct(self):
        # 1. Setup WebGL Bottleneck (0-2s)
        pipe = Rectangle(height=12, width=1.5, color=GREY_B, stroke_width=4)
        pipe.to_edge(LEFT, buff=1.5)
        
        webgl_circle = Circle(radius=0.6, color=RED, fill_opacity=0.3)
        webgl_text = Text("WebGL", font_size=32, weight=BOLD, color=RED)
        webgl_text.scale_to_fit_width(webgl_circle.get_width() * 0.8)
        webgl_node = VGroup(webgl_circle, webgl_text).move_to(pipe.get_center() + UP * 4)
        
        self.play(
            Create(pipe),
            FadeIn(webgl_node, shift=DOWN),
            run_time=2.0
        )

        # 2. Hitting a wall (2-5s)
        # The node struggles to move down the narrow pipe
        self.play(
            webgl_node.animate.shift(DOWN * 2),
            rate_func=slow_into,
            run_time=1.5
        )
        self.play(
            Indicate(webgl_node, color=RED_E, scale_factor=1.2),
            webgl_node.animate.set_color(RED_E),
            Flash(webgl_node, color=RED, flash_radius=0.8),
            run_time=1.5
        )

        # 3. WebGPU Takeover / Highway (5-8s)
        highway = Rectangle(height=14, width=7, color=GOLD_E, stroke_width=2, fill_opacity=0.1)
        webgpu_circle = Circle(radius=1.2, color=GOLD, fill_opacity=0.5)
        webgpu_text = Text("WebGPU", font_size=40, weight=BOLD, color=GOLD)
        webgpu_text.scale_to_fit_width(webgpu_circle.get_width() * 0.8)
        webgpu_node = VGroup(webgpu_circle, webgpu_text).move_to(ORIGIN)

        self.play(
            ReplacementTransform(pipe, highway),
            ReplacementTransform(webgl_node, webgpu_node),
            run_time=3.0,
            rate_func=rate_functions.ease_in_out_expo
        )

        # 4. Total Takeover - Expansion (8-11s)
        self.play(
            webgpu_node.animate.scale(1.5).set_color(TEAL),
            highway.animate.set_stroke(TEAL, width=8),
            run_time=3.0,
            rate_func=wiggle
        )

        # 5. Direct Access - Arrow (11-14s)
        access_arrow = Arrow(start=webgpu_node.get_right(), end=RIGHT * 4, color=WHITE, stroke_width=10)
        hardware_label = Text("HARDWARE", font_size=24, color=WHITE).next_to(access_arrow, UP)
        hardware_label.scale_to_fit_width(config.frame_width * 0.4)

        self.play(
            GrowArrow(access_arrow),
            Write(hardware_label),
            run_time=3.0
        )

        # 6. Final Wait (14-16.27s)
        self.wait(2.27)

class Segment001(Scene):
    def construct(self):
        # 1. Setup Stages (0-2s)
        input_stage = RoundedRectangle(height=2, width=3, color=BLUE, corner_radius=0.2)
        input_stage.to_corner(UL, buff=1.0)
        input_label = Text("INPUT", font_size=24).move_to(input_stage)
        
        compute_stage = RegularPolygon(n=6, color=PURPLE, fill_opacity=0.2).scale(2)
        compute_stage.move_to(ORIGIN)
        compute_label = Text("COMPUTE\nSHADER", font_size=28, weight=BOLD).move_to(compute_stage)
        compute_label.scale_to_fit_width(compute_stage.get_width() * 0.7)
        
        render_stage = RoundedRectangle(height=2, width=3, color=GREEN, corner_radius=0.2)
        render_stage.to_corner(DR, buff=1.0)
        render_label = Text("RENDER", font_size=24).move_to(render_stage)

        self.play(
            Create(input_stage), Write(input_label),
            Create(compute_stage), Write(compute_label),
            Create(render_stage), Write(render_label),
            run_time=2.0
        )

        # 2. Data flows in (2-5s)
        particles = VGroup(*[Dot(color=BLUE_A, radius=0.1) for _ in range(8)])
        particles.arrange_in_grid(rows=2, cols=4, buff=0.2).move_to(input_stage)

        self.play(
            LaggedStart(*[
                p.animate.move_to(compute_stage.get_center() + np.array([np.random.uniform(-0.5, 0.5), np.random.uniform(-0.5, 0.5), 0]))
                for p in particles
            ], lag_ratio=0.1),
            run_time=3.0
        )

        # 3. Massive Calculations - Morph (5-10s)
        stars = VGroup(*[Star(n=5, outer_radius=0.2, inner_radius=0.1, color=GOLD) for _ in range(8)])
        for i, s in enumerate(stars):
            s.move_to(particles[i].get_center())

        self.play(
            ReplacementTransform(particles, stars),
            stars.animate.rotate(TAU).scale(1.2),
            run_time=5.0,
            rate_func=smooth
        )

        # 4. Supercomputer Brain - Pulse (10-14s)
        rings = VGroup(*[Circle(radius=r, color=PURPLE, stroke_opacity=1-r/3) for r in np.linspace(0.5, 3, 5)])
        rings.move_to(compute_stage)

        self.play(
            LaggedStart(*[
                Succession(
                    FadeIn(ring, scale=0.5),
                    ring.animate.scale(1.5).set_stroke(opacity=0),
                ) for ring in rings
            ], lag_ratio=0.3),
            compute_stage.animate.set_fill(PURPLE, opacity=0.6),
            run_time=4.0
        )

        # 5. Every Pixel - Multiply (14-17s)
        pixel_grid = VGroup(*[Dot(radius=0.05, color=GREEN_A) for _ in range(64)])
        pixel_grid.arrange_in_grid(rows=8, cols=8, buff=0.1).move_to(render_stage)

        self.play(
            TransformFromCopy(stars, pixel_grid),
            stars.animate.set_opacity(0),
            run_time=3.0
        )

        # 6. Final Wait (17-19.03s)
        self.wait(2.03)

class Segment002(Scene):
    def construct(self):
        # 1. Setup GPU Core & AI Nodes (0-2s)
        gpu_core = RoundedRectangle(height=3, width=3, color=GOLD, corner_radius=0.5, fill_opacity=0.2)
        gpu_label = Text("WebGPU\nCORE", font_size=32, weight=BOLD, color=GOLD).move_to(gpu_core)
        gpu_label.scale_to_fit_width(gpu_core.get_width() * 0.8)
        
        ai_node_1 = Circle(radius=0.8, color=TEAL, fill_opacity=0.3).shift(UP * 4 + RIGHT * 2)
        ai_label_1 = Text("LLM", font_size=24).move_to(ai_node_1)
        
        ai_node_2 = Circle(radius=0.8, color=PURPLE, fill_opacity=0.3).shift(DOWN * 4 + RIGHT * 2)
        ai_label_2 = Text("Stable\nDiff", font_size=20).move_to(ai_node_2)

        self.play(
            DrawBorderThenFill(gpu_core), Write(gpu_label),
            FadeIn(ai_node_1, shift=LEFT), Write(ai_label_1),
            FadeIn(ai_node_2, shift=LEFT), Write(ai_label_2),
            run_time=2.0
        )

        # 2. Emit Particles (2-6s)
        def get_stream(start, end, color):
            return VGroup(*[
                Dot(radius=0.08, color=color).move_to(start)
                for _ in range(5)
            ])

        stream1 = get_stream(gpu_core.get_center(), ai_node_1.get_center(), TEAL)
        stream2 = get_stream(gpu_core.get_center(), ai_node_2.get_center(), PURPLE)

        self.play(
            LaggedStart(*[
                p.animate.move_to(ai_node_1.get_center())
                for p in stream1
            ], lag_ratio=0.2),
            LaggedStart(*[
                p.animate.move_to(ai_node_2.get_center())
                for p in stream2
            ], lag_ratio=0.2),
            run_time=4.0
        )

        # 3. Privacy Shield (6-10s)
        shield = SurroundingRectangle(gpu_core, color=WHITE, buff=0.3, corner_radius=0.6)
        shield_icon = Triangle(color=WHITE).scale(0.3).next_to(gpu_core, UP, buff=0.5)
        
        self.play(
            Create(shield),
            FadeIn(shield_icon, shift=DOWN),
            gpu_core.animate.set_fill(GREEN, opacity=0.4),
            run_time=4.0
        )

        # 4. Light Speed - Rapid Flashes (10-15s)
        lines = VGroup(
            Line(gpu_core.get_center(), ai_node_1.get_center(), color=GOLD_A, stroke_width=2),
            Line(gpu_core.get_center(), ai_node_2.get_center(), color=GOLD_A, stroke_width=2),
            Line(ai_node_1.get_center(), ai_node_2.get_center(), color=GOLD_A, stroke_width=1)
        )

        self.play(
            LaggedStartMap(ShowPassingFlash, lines, run_time=5.0, time_width=0.5, lag_ratio=0.1),
            ai_node_1.animate.scale(1.2),
            ai_node_2.animate.scale(1.2),
            rate_func=linear
        )

        # 5. Final Wait (15-17.86s)
        self.wait(2.86)

class Segment003(Scene):
    def construct(self):
        # 1. The Future Node (0-3s)
        future_circle = Circle(radius=1.5, color=WHITE, stroke_width=2)
        future_circle.set_fill(color=[BLUE, PURPLE, GOLD], opacity=0.3)
        future_text = Text("THE FUTURE", font_size=42, weight=BOLD)
        future_text.scale_to_fit_width(future_circle.get_width() * 0.8)
        future_node = VGroup(future_circle, future_text).move_to(UP * 2)

        glow = future_circle.copy().scale(1.2).set_stroke(width=0).set_fill(WHITE, opacity=0.1)

        self.play(
            SpinInFromNothing(future_node),
            FadeIn(glow, scale=0.8),
            run_time=3.0
        )

        # 2. Ultimate Platform - Expand (3-7s)
        platform_rect = RoundedRectangle(height=4, width=7, color=BLUE_D, fill_opacity=0.1)
        platform_rect.next_to(future_node, DOWN, buff=1)
        platform_label = Text("ULTIMATE PLATFORM", font_size=32, color=BLUE_A)
        platform_label.scale_to_fit_width(platform_rect.get_width() * 0.9)
        platform_label.move_to(platform_rect)

        self.play(
            future_node.animate.scale(1.2).shift(UP * 1),
            Create(platform_rect),
            Write(platform_label),
            run_time=4.0
        )

        # 3. Subscribe to Code Tapasya (7-11s)
        subscribe_badge = VGroup(
            RoundedRectangle(height=1.5, width=6, color=RED, fill_opacity=0.8, corner_radius=0.4),
            Text("SUBSCRIBE TO CODE TAPASYA", font_size=28, weight=BOLD, color=WHITE)
        )
        subscribe_badge[1].scale_to_fit_width(subscribe_badge[0].get_width() * 0.9)
        subscribe_badge.move_to(DOWN * 5)

        self.play(
            FadeIn(subscribe_badge, shift=UP),
            Wiggle(subscribe_badge, scale_value=1.1),
            Circumscribe(subscribe_badge, color=GOLD, time_width=2),
            run_time=4.0
        )

        # 4. Final Wait (11-14.88s)
        self.wait(3.88)
