class Segment000(Scene):
    def construct(self):
        # Narration: "Ever feel like your brain is a mess of 'what-ifs'? Decision trees are like a logical roadmap for your choices. They take messy, chaotic data and turn it into a clear, structured path. Let's break down how these digital brains actually represent knowledge!"
        
        # Chaotic Data
        dots = VGroup(*[Dot(point=[np.random.uniform(-4, 4), np.random.uniform(-3, 1), 0], color=BLUE_A, radius=0.05) for _ in range(50)])
        self.play(FadeIn(dots, shift=UP), run_time=2.0)
        
        # Root Node
        root_node = Circle(radius=0.6, color=GOLD).shift(UP * 2.5)
        root_text = Text("Decision", font_size=24).move_to(root_node.get_center())
        
        # Transform Chaos into Structure
        target_points = [
            UP*2.5, LEFT*2+UP*0.5, RIGHT*2+UP*0.5, 
            LEFT*3.5+DOWN*1.5, LEFT*0.5+DOWN*1.5, RIGHT*0.5+DOWN*1.5, RIGHT*3.5+DOWN*1.5
        ]
        
        path_lines = VGroup(
            Line(UP*2.5, LEFT*2+UP*0.5, color=GREY),
            Line(UP*2.5, RIGHT*2+UP*0.5, color=GREY),
            Line(LEFT*2+UP*0.5, LEFT*3.5+DOWN*1.5, color=GREY),
            Line(LEFT*2+UP*0.5, LEFT*0.5+DOWN*1.5, color=GREY)
        )

        self.play(
            ReplacementTransform(dots[:len(target_points)], VGroup(*[Dot(p, color=GOLD) for p in target_points])),
            FadeOut(dots[len(target_points):]),
            Create(root_node),
            Write(root_text),
            run_time=3.0
        )
        
        self.play(Create(path_lines), run_time=2.0)
        
        # SemanticBeat: root_node pulses
        self.play(Indicate(root_node, color=WHITE, scale_factor=1.2), run_time=1.5)
        
        # Title Text
        title = Text("DECISION TREES", weight=BOLD, color=BLUE_B).scale_to_fit_width(config.frame_width * 0.4).to_edge(DOWN)
        self.play(Write(title), run_time=1.5)
        
        self.play(root_node.animate.set_color(YELLOW), run_time=1.0)
        
        # Total time used: 2 + 3 + 2 + 1.5 + 1.5 + 1 = 11.0s
        self.wait(15.82 - 11.0)

class Segment004(Scene):
    def construct(self):
        # Narration: "Watch how the data moves! It starts at the root, undergoes a transformation at each internal stage, and flows down the branches based on logic. This sequential processing is what we call 'traversing the tree.'"
        
        root = Circle(radius=0.4, color=WHITE).shift(UP * 2.5)
        mid = Circle(radius=0.4, color=WHITE).shift(LEFT * 2 + UP * 0.5)
        leaf = Square(side_length=0.8, color=GREEN).shift(LEFT * 3 + DOWN * 2)
        
        path1 = Line(root.get_bottom(), mid.get_top(), buff=0.1)
        path2 = Line(mid.get_bottom(), leaf.get_top(), buff=0.1)
        
        tree = VGroup(root, mid, leaf, path1, path2)
        self.add(tree)
        
        walker = Dot(color=YELLOW, radius=0.15).move_to(UP * 4)
        
        # SemanticBeat: walker enters root node
        self.play(walker.animate.move_to(root.get_center()), run_time=1.5)
        self.play(Flash(root, color=YELLOW), run_time=1.0)
        
        # SemanticBeat: walker moves along path1 to mid_node
        self.play(MoveAlongPath(walker, path1), run_time=2.0)
        self.play(Indicate(mid), run_time=1.0)
        
        self.play(MoveAlongPath(walker, path2), run_time=2.0)
        self.play(walker.animate.scale(2).set_color(GREEN), run_time=1.0)
        
        label = Text("TRAVERSING", color=BLUE).scale_to_fit_width(4).to_edge(RIGHT)
        self.play(Write(label), run_time=1.5)
        
        # Total: 1.5 + 1 + 2 + 1 + 2 + 1 + 1.5 = 10.0s
        self.wait(13.10 - 10.0)

class Segment008(Scene):
    def construct(self):
        # Narration: "Sometimes a tree gets too messy—this is called overfitting. To fix it, we use 'Pruning.' We literally cut back unnecessary branches to make the tree simpler and more accurate for new, unseen data."
        
        # Messy Tree
        root = Dot(UP*2, color=WHITE)
        level1 = VGroup(Dot(UP*0.5 + LEFT*2), Dot(UP*0.5 + RIGHT*2))
        level2 = VGroup(*[Dot(DOWN*1 + LEFT*3 + RIGHT*i) for i in range(7)])
        lines = VGroup(*[Line(root.get_center(), l1.get_center(), stroke_width=2) for l1 in level1])
        for l1 in level1:
            for l2 in level2:
                lines.add(Line(l1.get_center(), l2.get_center(), stroke_width=1, color=RED_E))
        
        complex_tree = VGroup(root, level1, level2, lines)
        
        # SemanticBeat: complex_tree pulses red
        self.play(Create(complex_tree), run_time=3.0)
        self.play(Indicate(complex_tree, color=RED), run_time=1.5)
        
        overfit_text = Text("OVERFITTING!", color=RED, weight=BOLD).to_edge(UP)
        self.play(Write(overfit_text), run_time=1.0)
        
        # SemanticBeat: outer nodes fade away (Pruning)
        pruning_tool = Text("✂️ PRUNING", font_size=48, color=GOLD).shift(RIGHT*4)
        self.play(FadeIn(pruning_tool, shift=LEFT), run_time=1.5)
        
        self.play(
            FadeOut(level2),
            lines[2:].animate.set_stroke(opacity=0),
            overfit_text.animate.become(Text("OPTIMIZED", color=GREEN).to_edge(UP)),
            run_time=3.0
        )
        
        self.play(complex_tree[0:2].animate.set_color(GREEN), run_time=1.0)
        
        # Total: 3 + 1.5 + 1 + 1.5 + 3 + 1 = 11.0s
        self.wait(13.15 - 11.0)
